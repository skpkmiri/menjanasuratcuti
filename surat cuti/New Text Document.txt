        function formatMalayDate(dateString) {
            if (!dateString) return '__________';
            const date = new Date(dateString);
            if (isNaN(date.getTime())) return '__________';
            const months = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember'];
            return `${date.getDate()} ${months[date.getMonth()]} ${date.getFullYear()}`;

        // Parse CSV Helper
        function parseRobustCSV(text) {
            let p = '', row = [''], ret = [row], i = 0, r = 0, s = !0, l;
            for (l of text) {
                if ('"' === l) { if (s && l === p) row[i] += l; s = !s; } 
                else if (',' === l && s) l = row[++i] = '';
                else if ('\n' === l && s) { if ('\r' === p) row[i] = row[i].slice(0, -1); row = ret[++r] = ['']; i = 0; } 
                else row[i] += l;
                p = l;
            }
            return ret;
        }

        function getCoords(e) {
            const rect = canvas.getBoundingClientRect();
            const clientX = e.clientX || (e.touches && e.touches[0].clientX);
            const clientY = e.clientY || (e.touches && e.touches[0].clientY);

            if (!nama || !tujuan || !tMula || !tTamat) {
                return showToast("Sila lengkapkan semua butiran borang!", "warning");
            }

            if (db) {
                try {
                    const rekodBaru = {
                        nama: nama,
                        ic: activeUserIC,
                        jawatan: document.getElementById('input-jawatan').value,
                        jenisCuti: document.getElementById('input-jeniscuti').value,
                        tujuan: tujuan,
                        tarikhMula: tMula,
                        tarikhTamat: tTamat,
                        hari: parseInt(document.getElementById('input-hari').value),
                        tarikhPermohonan: document.getElementById('input-tarikhmohon').value,
                        sertakanKelulusan: document.getElementById('input-sertakan-kelulusan').checked,
                        signatureData: signatureData,
                        createdAt: Date.now()
                    };

                    if (editingId) {
                        await updateDoc(doc(db, 'artifacts', appId, 'public', 'data', 'rekod_cuti_skpk', editingId), rekodBaru);
                    } else {
                        const ref = await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'rekod_cuti_skpk'), rekodBaru);
                        editingId = ref.id;
                        document.getElementById('edit-mode-alert').classList.remove('hidden');
                        document.getElementById('btn-save-download-text').textContent = 'Kemaskini & Muat Turun';
                    }
                    showToast("Tersimpan ke pangkalan data.", "success");
                } catch (err) {
                    console.error(err);
                    showToast("Ralat menyimpan ke DB.", "error");
                }
            }

            // PDF Generation
            const element = document.getElementById('surat-preview-pdf');
            if (typeof html2pdf !== 'undefined') {
                const opt = {
                    margin:       15,
                    filename:     `Cuti_${nama.replace(/\s+/g, '_')}_${document.getElementById('input-jeniscuti').value.replace(/\s+/g, '_')}.pdf`,
                    image:        { type: 'jpeg', quality: 0.98 },
                    html2canvas:  { scale: 2, useCORS: true, letterRendering: true },
                    jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
                };
                html2pdf().from(element).set(opt).save().catch(err => {
                    console.error(err); showToast("Ralat PDF.", "error");
                });
            } else {
                showToast("Pemacu PDF sedang dimuatkan...", "warning");
            }
        });

        // Initialize App
        fetchCSV();
        initFirebase();
        updatePreview();

    </script>
